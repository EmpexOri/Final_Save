namespace = songhay_collapse

country_event = {
    id = songhay_collapse.1
    title = "songhay_collapse.1.t"
    desc = "songhay_collapse.1.d"
    picture = DISASTER_eventPicture
    fire_only_once = yes
    major = yes
    
    trigger = {
        tag = SON
        is_year = 1645
        NOT = { is_year = 1701 }
        NOT = { has_country_flag = songhay_collapsed }
        total_development = 100
    }
    
    mean_time_to_happen = { months = 12 }
    
    immediate = {
        set_country_flag = songhay_collapsed
        country_event = { id = songhay_collapse.2 days = 1825 }  # 5 years
    }
    
    option = {
        name = "songhay_collapse.1.a"
        
        add_stability = -3
        
        add_country_modifier = {
            name = songhay_internal_collapse
            duration = 10950  # 30 years
        }
        
        every_owned_province = {
            limit = { NOT = { is_capital = yes } }
            add_unrest = 20
            add_province_modifier = {
                name = songhay_revolt
                duration = 10950
            }
        }
        
        every_owned_province = {
            limit = { NOT = { is_capital = yes } }
            spawn_rebels = {
                type = particularist
                size = 12
            }
        }
        
        capital_scope = {
            spawn_rebels = {
                type = pretender
                size = 15
            }
        }
    }
}

			country_event = { id = ilkhanate_reform.2 days = 1 }
		}
	}
