namespace = patch_china_events

#Kurokawa Kinzan
#Base game defines trade good in Kai as gold starting from 1498, but this doesn't seem to work
province_event = {
	id = patch_china_events.0
	
	title = patch_china_events.0.t
	desc = patch_china_events.0.d

	picture = COURT_eventPicture
	
	fire_only_once = yes
	
	trigger = {
			AND = { 
				province_id = 1029 # Kai
				is_year = 1498
				NOT = { trade_goods = gold }
			}
	}
	
	mean_time_to_happen = {
		months = 200
	}
	
	option = { 
		name = patch_china_events.0.a
		change_trade_goods = gold
	}
	
}


country_event = { #The Decadent State of the Ming Dynasty
	id = patch_china_events.1
	title = patch_china_events.1.t
	desc = patch_china_events.1.d
	picture = COUNTRY_COLLAPSE_eventPicture
	fire_only_once = yes 
	trigger = {
		has_dlc = "Domination"
		tag = CHN
		is_emperor_of_china = yes 
		NOT = { has_country_flag = mng_received_starting_event } 
	}
	is_triggered_only = yes 
	
	mean_time_to_happen = { days = 1 } 
	
	immediate = {
		hidden_effect = {
			add_country_modifier = { 
				name = mng_starting_malus_5_modifier
				duration = -1 
			}
			set_country_flag = mng_received_starting_event
		}
	}
	option = {
		name = patch_china_events.1.a 
		
		custom_tooltip = mng_great_autonomy_tt
	}
}
