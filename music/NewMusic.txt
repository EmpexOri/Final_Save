song = {
    name = cupofgold
    chance = {
        # Base chance: 1 only if the country is China, Japan, or Korea
        modifier = {
            factor = 1
            OR = {
                tag = CHN
                tag = JAP   # Japan
                tag = KOR   # Korea
            }
        }
        # For all other countries, force the chance to 0
        modifier = {
            factor = 0
            NOT = {
                OR = {
                    tag = CHN
                    tag = JAP
                    tag = KOR
                }
            }
        }
        # Double the chance when at war (applies on top of the base)
        modifier = {
            factor = 2
            is_at_war = yes
        }
    }
}
